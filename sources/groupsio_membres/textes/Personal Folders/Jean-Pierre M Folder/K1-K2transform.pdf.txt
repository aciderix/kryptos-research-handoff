def count_transformations(decoded, encoded): 
    transformation_dict = {} 
     
    for d, e in zip(decoded, encoded): 
        if d not in transformation_dict: 
            transformation_dict[d] = {} 
        if e not in transformation_dict[d]: 
            transformation_dict[d][e] = 0 
        transformation_dict[d][e] += 1 
    return transformation_dict 
def sort_transformations(transformations, custom_alphabet): 
    sorted_transformations = {} 
    for d in custom_alphabet: 
        if d in transformations: 
            sorted_transformations[d] = dict(sorted(transformations[d].items(), key=lambda x: custom_alphabet.index(x[0]))) 
    return sorted_transformations 
def print_transformations(decoded, encoded, sorted_transformations): 
    for d, encoded_letters in sorted_transformations.items(): 
        transformations_str = ', '.join([f"{e}: {count}" for e, count in encoded_letters.items()]) 
        occurrences = [e for d_char, e in zip(decoded, encoded) if d_char == d] 
        print(f"{d}: {transformations_str} ( {' '.join(occurrences)} )")
